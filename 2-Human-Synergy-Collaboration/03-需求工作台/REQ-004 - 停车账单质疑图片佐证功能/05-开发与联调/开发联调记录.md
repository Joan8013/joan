@DisplayName("同一客户对同一账单 24 小时内二次质疑应幂等")
class DisputeOrderServiceIT extends AbstractMySqlIT {

    @Autowired DisputeOrderService service;
    @Autowired DisputeOrderMapper mapper;

    @Test
    void 重复质疑24h内应幂等() {
        DisputeOrderCreateReq req = DisputeReqBuilder.aReq()
                .customerNo("C1").billNo("B1").parkingLotNo("P1")
                .amount("100.00").build();

        DisputeResult first = service.create(req);   // 首次建单
        DisputeResult second = service.create(req);  // 24h 内重复

        assertThat(second.getCode()).isEqualTo("DUPLICATE");
        assertThat(mapper.countByCustomerAndBillWithin24h("C1", "B1")).isEqualTo(1);
    }
}
```

### 绿：最小实现（示意）

// DisputeOrderServiceImpl.create()
public DisputeResult create(DisputeOrderCreateReq req) {
    // 24h 窗口幂等预校验（A1: 行锁串行化 + 预校验）
    if (mapper.countByCustomerAndBillWithin24h(req.getCustomerNo(), req.getBillNo()) > 0) {
        return DisputeResult.duplicate();   // code=DUPLICATE，不建单
    }
    // ... 正常建单（TASK-003 负责字段映射）
}
```

    where customer_no = #{customerNo} and bill_no = #{billNo}
      and dispute_time >= date_sub(now(), interval 24 hour)
</select>
