Private Declare Function md5_string Lib "C:\Users\bao\Desktop\tmp\vb\MyAESDLL.dll" (ByVal arg1 As String) As String
Private Declare Function base64_encode Lib "C:\Users\bao\Desktop\tmp\vb\MyAESDLL.dll" (ByVal arg1 As String, ByVal arg2 As Long) As String
Private Declare Function aes_encode Lib "C:\Users\bao\Desktop\tmp\vb\MyAESDLL.dll" (ByVal arg1 As String, ByVal arg2 As String) As String
Private Declare Function aes_decode Lib "C:\Users\bao\Desktop\tmp\vb\MyAESDLL.dll" (ByVal arg1 As String, ByVal arg2 As String) As String
Private Declare Function freepointer Lib "C:\Users\bao\Desktop\tmp\vb\MyAESDLL.dll" (ByVal arg1 As String) As Long



Private Sub Command1_Click()
    Dim s As String
    
    s = "aaaa"
    
    'MD5算法：
    enc = md5_string(s)
    Call freepointer(enc)
    MsgBox "md5结果：" & enc
    
    'Base64 Encode
    base64 = base64_encode(enc, Len(enc))
    Call freepointer(base64)
    MsgBox "base64编码结果：" & base64
    
    'aes 加密:
    aesen = aes_encode(enc, "1234567890123456")
    Call freepointer(aesen)
    MsgBox "aes 加密结果：" & aesen
    
    'aes 解码:
    aesdec = aes_decode(aesen, "1234567890123456")
    Call freepointer(aesdec)
    MsgBox "aes 解密结果：" & aesdec
    
End Sub
